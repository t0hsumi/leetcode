# step1
考えたこと
- めちゃくちゃ単純にやれば、先頭文字から始めて、各文字に対して後続文字を全部調べればできる。
- 文字列の長さをnとして、time complexityがO(n^2)なので、O(n)にしたい。
- 文字列を後ろからせめるとどうか
    - ある文字が始めて登場したら、min_indexを更新
    - 登場したことのある文字に対しては特に処理せず
    - ある文字が2回目に出現したら、後続文字のなかから、条件の満たすものを探してmin_indexを更新
        - この作業があるため結局O(n^2)相当と考えられる。（英語小文字のみなので、ここまでひどくはないだろうが）
- 最初にiterateしてindexを記録する. O(n)
    - 入力条件的に英語小文字だけしかこないので、それらに対して全てチェック
    - もしくは、もう一度、iterateして最小のindexを探す。

```python
from collections import defaultdict
import string


class Solution:
    def firstUniqChar(self, s: str) -> int:
        char_to_indices = defaultdict(list)
        for index, char in enumerate(s):
            char_to_indices[char].append(index)
        min_non_repeating_index = len(s)
        for lowercase in string.ascii_lowercase:
            if len(char_to_indices[lowercase]) == 1:
                min_non_repeating_index = min(
                    min_non_repeating_index,
                    char_to_indices[lowercase][0]
                )
        if min_non_repeating_index == len(s):
            return -1
        return min_non_repeating_index
```

```python
from collections import defaultdict


class Solution:
    def firstUniqChar(self, s: str) -> int:
        char_to_indices = defaultdict(list)
        for index, char in enumerate(s):
            char_to_indices[char].append(index)
        for index, char in enumerate(s):
            if len(char_to_indices[char]) == 1:
                return index
        return -1
```
こっちの方が最初書いたものよりはみやすい。
入力条件によっては（めちゃくちゃ長い文字列の最後だけuniqueとか）、こっちの方が遅くなりそう。

こっちは、入力条件に依存しないで、文字列に数字とかきても動くのがいい。

LRUcacheを使った解法
```pythonclass ListNode:
    def __init__(self, value=0, prev=None, next_node=None):
        self.value = value
        self.prev = prev
        self.next = next_node


class Deque:
    def __init__(self):
        self.dummy_head = ListNode()
        self.dummy_tail = ListNode()
        self.dummy_head.next = self.dummy_tail
        self.dummy_tail.prev = self.dummy_head

    def append(self, value) -> None:
        node = ListNode(value=value)
        next_node = self.dummy_tail
        prev_node = self.dummy_tail.prev

        node.next = next_node
        node.prev = prev_node

        next_node.prev = node
        prev_node.next = node

    def popleft(self):
        node = self.dummy_head.next
        return self._remove(node).value

    def remove(self, value) -> ListNode:
        node = self.dummy_head.next
        while node != self.dummy_tail:
            if node.value == value:
                return self._remove(node)
            node = node.next
        raise IndexError(
            "Deque.remove(): value is not in the deque",
            f"value = {value}"
        )

    def _remove(self, node: ListNode) -> ListNode:
        next_node = node.next
        prev_node = node.prev

        next_node.prev = prev_node
        prev_node.next = next_node

        node.next = None
        node.prev = None
        return node


class LRUCache:
    def __init__(self):
        self.access_history = Deque()
        self.cache = {}

    def get(self, key):
        if key not in self.cache:
            return -1
        self.access_history.remove(key)
        self.access_history.append(key)
        return self.cache[key]

    def put(self, key, value) -> None:
        if key in self.cache:
            self.access_history.remove(key)
        self.cache[key] = value
        self.access_history.append(key)
    
    def evict(self, key):
        if key not in self.cache:
            raise KeyError(
                "LRUCache.evict(): No value found: "
                f"key = {key}"
            )
        value = self.cache[key]
        del self.cache[key]
        self.access_history.remove(key)
        return value
    
    def evict_oldest(self):
        if self.empty():
            raise KeyError(
                "LRUCache.evict_oldest(): LRUCache is empty"
            )
        oldest_key = self.access_history.popleft()
        oldest_value = self.cache[oldest_key]
        del self.cache[oldest_key]
        return oldest_value

    def empty(self):
        return not self.cache


class Solution:
    def firstUniqChar(self, s: str) -> int:
        unique_char_index = LRUCache()
        repeating_chars = set()
        for index, char in enumerate(s):
            if char in repeating_chars:
                continue
            if unique_char_index.get(char) != -1:
                unique_char_index.evict(char)
                repeating_chars.add(char)
                continue
            unique_char_index.put(char, index)
        if unique_char_index.empty():
            return -1  # Not Found
        return unique_char_index.evict_oldest()
```
