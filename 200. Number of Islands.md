class Solution:
    def numIslands(self, grid: List[List[str]]) -> int:
        def visit_island(
            visited: List[List[bool]],
            row: int,
            column: int
        ) -> None:
            directions = (
                (1, 0),
                (-1, 0),
                (0, 1),
                (0, -1),
            )
            next_positions = [(row, column)]
            while next_positions:
                row, column = next_positions.pop()
                visited[row][column] = True
                for dy, dx in directions:
                    new_row = row + dy
                    new_column = column + dx
                    if (
                        0 <= new_row < len(grid)
                        and 0 <= new_column < len(grid[0])
                        and grid[new_row][new_column] == '1'
                        and not visited[new_row][new_column]
                    ):
                        next_positions.append((new_row, new_column))
        visited = [[False] * len(grid[0]) for _ in range(len(grid))]
        num_of_islands = 0
        for i in range(len(grid)):
            for j in range(len(grid[0])):
                if grid[i][j] == '1' and not visited[i][j]:
                    visit_island(visited, i, j)
                    num_of_islands += 1
        return num_of_islands

class Solution:
    def numIslands(self, grid: List[List[str]]) -> int:
        NUM_ROW = len(grid)
        NUM_COLUMN = len(grid[0])
        LAND = '1'
        WATER = '0'

        def traverse_island(
            visited: List[List[bool]],
            row: int,
            column: int
        ) -> None:
            directions = (
                (1, 0),
                (-1, 0),
                (0, 1),
                (0, -1),
            )
            next_positions = [(row, column)]
            while next_positions:
                row, column = next_positions.pop()
                visited[row][column] = True
                for delta_row, delta_column in directions:
                    next_row = row + delta_row
                    next_column = column + delta_column
                    if (
                        not 0 <= next_row < NUM_ROW
                        or not 0 <= next_column < NUM_COLUMN
                    ):
                        continue
                    if grid[next_row][next_column] == WATER:
                        continue
                    if visited[next_row][next_column]:
                        continue
                    next_positions.append((next_row, next_column))

        num_of_islands = 0
        visited = [[False] * NUM_COLUMN for _ in range(NUM_ROW)]
        for row in range(NUM_ROW):
            for column in range(NUM_COLUMN):
                if grid[row][column] == LAND and not visited[row][column]:
                    traverse_island(visited, row, column)
                    num_of_islands += 1
        return num_of_islands

UnionFind
```python
class UnionFind:
    def __init__(self, elements):
        self.parent = {element: element for element in elements}
        self.rank = {element: 1 for element in elements}
        self.count = len(elements)

    def find(self, element):
        while element != self.parent[element]:
            self.parent[element] = self.parent[self.parent[element]]
            element = self.parent[element]
        return element

    def union(self, element1, element2):
        root1 = self.find(element1)
        root2 = self.find(element2)
        
        if root1 == root2:
            return
        
        if self.rank[root1] > self.rank[root2]:
            self.parent[root2] = root1
        elif self.rank[root1] < self.rank[root2]:
            self.parent[root1] = root2
        else:
            self.parent[root1] = root2
            self.rank[root1] += 1

        self.count -= 1


class Solution:
    def numIslands(self, grid: List[List[str]]) -> int:
        NUM_ROW = len(grid)
        NUM_COLUMN = len(grid[0])
        LAND = '1'
        points = set()
        for row in range(NUM_ROW):
            for column in range(NUM_COLUMN):
                if grid[row][column] == LAND:
                    points.add((row, column))
        uf = UnionFind(points)

        for row, column in points:
            for delta_row, delta_column in [(0, 1), (1, 0)]:
                next_row = row + delta_row
                next_column = column + delta_column
                if (next_row, next_column) in points:
                    uf.union((row, column), (next_row, next_column))
        return uf.count
```

# step 3
traverse_island内では、ネストを深くしないためにcontinueを多用した。
step1のやり方とは変更をした。が、step1では次に探索すべき条件が列挙されているのに対し、
step2以降では探索をやめるべき条件を弾いたものを探索している感じになっている。
問題設定が簡単なのでstep2以降の書き方でもいい気がするが、water, land以外に新たな敵国とかでてきたりして
めんどくさくなった時、step2以降の書き方だと、結局何を探索しようとしているのかわからなくなりそうだな
と感じた。

結局、directionsとしてまとめることにした。next_row, next_columnを使って、探索条件を同じように
チェックできるためである。探索条件を書く場所を関数冒頭にすればベタ書きでも問題ないが、
再帰を減らそうとすると、再帰関数の呼び出し直前にチェックする必要があり、その時にベタ書きだと不便に感じた。

        for row, column in points:
            if (row + 1, column) in points:
                uf.union((row, column), (row + 1, column))
            if (row, column + 1) in points:
                uf.union((row, column), (row, column + 1))
